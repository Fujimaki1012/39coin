/** * ハトメッセ (Hato-Messe) Backend v1.9.3 * 2026/04/06 更新 * * 修正内容: * - 応答の確実性を向上（Failed to fetch対策） * - 送信者名取得のロジック修正 */ const APP_URL = "https://fujimaki1012.github.io/39coin/39coin.html"; function doGet(e) { try { if (!e || !e.parameter) return ContentService.createTextOutput("API Alive"); const action = e.parameter.action; const email = e.parameter.email ? e.parameter.email.trim().toLowerCase() : ""; const nfcId = e.parameter.nfcId || ""; if (action === 'autoLogin') return loginAndGetFullData(email, null, true); if (action === 'register') return processRegistrationAPI(email, nfcId); if (action === 'getUserData') return loginAndGetFullData(email, nfcId); if (action === 'sendPoint') return sendPointByHato(email, e.parameter.targetMode, e.parameter.targetValue, e.parameter.message); if (action === 'getEmployeeList') return getEmployeeList(); } catch (err) { return createJsonResponse({ error: "システムエラー: " + err.message }); } } function getEmployeeList() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const data = ss.getSheetByName('社員マスタ').getDataRange().getValues(); const list = data.slice(1).map(row => row[0]).filter(name => name); return createJsonResponse({ success: true, list: list }); } function loginAndGetFullData(email, nfcId, isAuto = false) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const masterSheet = ss.getSheetByName('社員マスタ'); const userData = masterSheet.getDataRange().getValues(); const historyData = ss.getSheetByName('履歴').getDataRange().getValues(); let currentUser = null; let userRowIndex = -1; const userMap = {}; for (let i = 1; i < userData.length; i++) { const row = userData[i]; const rowEmail = String(row[1]).trim().toLowerCase(); if (rowEmail) userMap[rowEmail] = row[0]; if (rowEmail === email) { if (isAuto || nfcId === String(row[2]).trim() || nfcId === String(row[3]).trim()) { currentUser = { name: row[0], email: rowEmail, remains: row[4], received: row[5], welcome: row[6] }; userRowIndex = i + 1; } } } if (!currentUser) return createJsonResponse({ error: "認証失敗：登録済みのカードではありません。" }); if (!currentUser.welcome) { sendChatToUser(currentUser.email, "ハトメッセへようこそ！🕊️\nこれから、たくさんの感謝をハトに乗せて飛ばしましょう！", currentUser.name); masterSheet.getRange(userRowIndex, 7).setValue(new Date()); } let receivedLogs = []; let sentLogs = []; for (let i = historyData.length - 1; i >= 1; i--) { const row = historyData[i]; const log = { sender: userMap[String(row[1]).toLowerCase()] || "不明", receiver: userMap[String(row[2]).toLowerCase()] || "不明", text: row[4] || "", date: Utilities.formatDate(new Date(row[0]), "JST", "MM/dd HH:mm") }; if (String(row[2]).toLowerCase() === email) receivedLogs.push(log); if (String(row[1]).toLowerCase() === email) sentLogs.push(log); } return createJsonResponse({ success: true, user: currentUser, received: receivedLogs, sent: sentLogs }); } function sendPointByHato(fromEmail, mode, targetValue, message) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const userData = ss.getSheetByName('社員マスタ').getDataRange().getValues(); const historySheet = ss.getSheetByName('履歴'); let targetEmail = ""; let targetName = ""; let senderName = "誰か"; for (let i = 1; i < userData.length; i++) { const rowEmail = String(userData[i][1]).toLowerCase(); if (rowEmail === fromEmail.toLowerCase()) senderName = userData[i][0]; if (mode === 'touch') { if (String(userData[i][2]) === targetValue || String(userData[i][3]) === targetValue) { targetName = userData[i][0]; targetEmail = rowEmail; } } else if (userData[i][0] === targetValue) { targetName = userData[i][0]; targetEmail = rowEmail; } } if (!targetEmail) return createJsonResponse({ error: "送信先が見つかりません" }); if (targetEmail === fromEmail.toLowerCase()) return createJsonResponse({ error: "自分には飛ばせません" }); // シートへの記録を優先実行 historySheet.appendRow([new Date(), fromEmail, targetEmail, 1, message]); // 通知処理。エラーが起きても無視してレスポンスへ進む try { sendChatToUser(targetEmail, `${senderName} さんからハトメッセが届きました！🕊️`, targetName); } catch (e) { console.log("Chat notification failed but ignored: " + e.message); } return createJsonResponse({ success: true, receiver: targetName }); } function sendChatToUser(targetEmail, text, targetName) { const message = { "text": text, "cardsV2": [{ "cardId": "hatoNotification", "card": { "header": { "title": "🕊️ ハトメッセが届きました", "subtitle": targetName + " さんへ" }, "sections": [{ "widgets": [{ "buttonList": { "buttons": [{ "text": "アプリで確認", "onClick": { "openLink": { "url": APP_URL + "?email=" + targetEmail } } }] } }] }] } }] }; Chat.Spaces.Messages.create(message, "spaces/app_direct_messages", { "messageReplyOption": "REPLY_MESSAGE_FALLBACK" }); } function processRegistrationAPI(email, nfcId) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('社員マスタ'); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][2]).trim() === nfcId || String(data[i][3]).trim() === nfcId) { return createJsonResponse({ error: "このカードは既に登録済みです。" }); } } for (let i = 1; i < data.length; i++) { if (String(data[i][1]).trim().toLowerCase() === email) { const col = data[i][2] ? 4 : 3; sheet.getRange(i + 1, col).setValue(nfcId); return createJsonResponse({ success: true, message: "✅ カード登録完了！" }); } } return createJsonResponse({ error: "アドレスが見つかりません。" }); } function createJsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
